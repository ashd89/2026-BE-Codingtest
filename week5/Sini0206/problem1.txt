import java.io.*;
import java.util.*;

class Main {
    static int[] table;
    static String[] nicknames;
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        
        StringTokenizer st = new StringTokenizer(br.readLine());
        int N = Integer.parseInt(st.nextToken()); // 10^5
        int M = Integer.parseInt(st.nextToken()); // 10^5 -> O(MlogN)
        
        table = new int[N];
        nicknames = new String[N];
        for (int i=0; i<N; i++) {
            st = new StringTokenizer(br.readLine());
            String nickname = st.nextToken();
            int power = Integer.parseInt(st.nextToken());
            table[i] = power;
            nicknames[i] = nickname;
        }
        
        for (int i=0; i<M; i++) {
            st = new StringTokenizer(br.readLine());
            int power = Integer.parseInt(st.nextToken());
            bw.write(searchNickname(power, 0, N-1));
            bw.newLine();
        }
        bw.flush();
        bw.close();     
    }
    
    public static String searchNickname(int n, int left, int right) {
        int mid = (left+right)/2; 
            
        if (n <= table[mid]) {
            if (mid == 0 || n>table[mid-1])
                return nicknames[mid];  // 찾음
            else
                return searchNickname(n, left, mid-1);                                  
        }
        else
            return searchNickname(n, mid+1, right);
    }    
   
}
